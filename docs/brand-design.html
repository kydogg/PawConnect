Swift UI Implementation

// MARK: - Typography Extensions
extension Font {
    // Headlines - SF Pro Display
    static let appDisplayLarge = Font.system(size: 34, weight: .bold, design: .default)
    static let appHeadingH1 = Font.system(size: 24, weight: .bold, design: .default)
    static let appHeadingH2 = Font.system(size: 20, weight: .semibold, design: .default)

    // Body Text - SF Pro Text
    static let appBodyLarge = Font.system(size: 17, weight: .regular, design: .default)
    static let appBodyRegular = Font.system(size: 15, weight: .regular, design: .default)
    static let appBodySmall = Font.system(size: 13, weight: .regular, design: .default)

    // UI Elements
    static let appButtonLarge = Font.system(size: 17, weight: .semibold, design: .default)
    static let appButtonSmall = Font.system(size: 15, weight: .medium, design: .default)
    static let appCaption = Font.system(size: 12, weight: .regular, design: .default)
}

// Usage in SwiftUI Views
Text("Welcome Home")
    .font(.appDisplayLarge)
    .foregroundColor(AppColor.Primary.sunset)
